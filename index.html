const SHEET_INSCRIPTOS = "Inscriptos"; const SHEET_ABANDONADOS = "Carrito abandonado"; const INSCRIPTOS_HEADERS = ["Email", "Nombre", "Celular", "Tipo de asistente", "Ítem", "Tipo", "Monto", "ID de pago MP", "Código QR", "Fecha", "Acreditado"]; const ABANDONADOS_HEADERS = ["Cart Ref", "Email", "Nombre", "Teléfono", "Tipo de asistente", "Ítems", "Monto total", "Fecha", "Convertido"]; function jsonOut(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // Esto SIEMPRE fuerza la fila 1 a los encabezados correctos, exista o no la // hoja de antes. Así, si la pestaña ya existía con encabezados viejos (como // pasó con "Inscriptos", que es de mucho antes de este esquema), se corrige // sola en vez de quedar desalineada para siempre en silencio. function getSheetWithHeaders(name, headers) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); } sheet.getRange(1, 1, 1, headers.length).setValues([headers]); return sheet; } function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (data.action === "checkin") { return handleCheckin(data); } if (data.action === "abandoned_cart_add") { return handleAbandonedAdd(data); } if (data.action === "mark_converted") { return handleMarkConverted(data); } // ACCIÓN DEFAULT: pago confirmado — una fila por cada ítem comprado // (Día 1 y/o cada taller), siempre en la hoja "Inscriptos" por nombre // explícito (nunca getActiveSheet()). return handleInsertInscriptos(data); } catch (err) { return jsonOut({ status: "error", message: String(err) }); } } // ---- Pago confirmado: una fila por ítem comprado ---- function handleInsertInscriptos(data) { const sheet = getSheetWithHeaders(SHEET_INSCRIPTOS, INSCRIPTOS_HEADERS); const items = Array.isArray(data.items) && data.items.length ? data.items : [{ nombre: data.ticket_type || "Entrada general", tipo: "General", monto: data.monto || "" }]; items.forEach(function (item) { sheet.appendRow([ data.email || "", data.nombre || "", data.telefono || "", data.tipo_asistente || "", item.nombre || "", item.tipo || "", item.monto || "", data.mp_payment_id || "", data.qr_code || "", data.fecha || new Date().toISOString(), "" // Acreditado (vacío al principio) ]); }); return jsonOut({ status: "ok", action: "insert", rows: items.length }); } // ---- Acreditación: marca TODAS las filas con ese código QR (una persona // puede tener varias filas si compró varias cosas) ---- function handleCheckin(data) { const sheet = getSheetWithHeaders(SHEET_INSCRIPTOS, INSCRIPTOS_HEADERS); const values = sheet.getDataRange().getValues(); const qrColIndex = 8; // columna I = "Código QR" (0-based; se corrió por Celular + Tipo de asistente) const acreditadoColIndex = 10; // columna K = "Acreditado" (0-based) let matches = 0; for (let i = 1; i < values.length; i++) { if (values[i][qrColIndex] === data.qr_code) { sheet.getRange(i + 1, acreditadoColIndex + 1).setValue( data.checked_in_at || new Date().toISOString() ); matches++; } } return jsonOut({ status: matches > 0 ? "ok" : "not_found", action: "checkin", rows: matches }); } // ---- Carrito abandonado: se registra apenas alguien elige forma de pago // (Mercado Pago o Transferencia) en el checkout, ANTES de completar el // pago. Es idempotente por cart_ref: si ya existe una fila con ese mismo // carrito, la actualiza en el lugar en vez de duplicarla — así se puede // llamar varias veces a medida que hay más datos disponibles (por ejemplo, // primero con los datos que se van tipeando, después con el monto final ya // calculado) sin ensuciar la hoja con filas repetidas. function handleAbandonedAdd(data) { const sheet = getSheetWithHeaders(SHEET_ABANDONADOS, ABANDONADOS_HEADERS); const refColIndex = 0; const convertidoColIndex = 8; const cartRef = data.cart_ref || ""; if (cartRef) { const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (values[i][refColIndex] === cartRef) { const rowNumber = i + 1; sheet.getRange(rowNumber, 1, 1, 8).setValues([[ cartRef, data.email || values[i][1] || "", data.nombre || values[i][2] || "", data.telefono || values[i][3] || "", data.tipo_asistente || values[i][4] || "", data.items || values[i][5] || "", data.monto_total || values[i][6] || "", data.fecha || new Date().toISOString(), ]]); return jsonOut({ status: "ok", action: "abandoned_cart_add", updated: true, row: rowNumber }); } } } sheet.appendRow([ cartRef, data.email || "", data.nombre || "", data.telefono || "", data.tipo_asistente || "", data.items || "", data.monto_total || "", data.fecha || new Date().toISOString(), "" // Convertido (vacío = todavía no pagó) ]); return jsonOut({ status: "ok", action: "abandoned_cart_add", updated: false }); } // ---- Marca como convertido un carrito abandonado cuando el pago se // confirma. Busca primero por cart_ref (lo llama mp-webhook); si no viene // o no matchea, busca por email (lo llama admin-action al cargar un pago // manual, por ejemplo una transferencia confirmada por WhatsApp) ---- function handleMarkConverted(data) { const sheet = getSheetWithHeaders(SHEET_ABANDONADOS, ABANDONADOS_HEADERS); const values = sheet.getDataRange().getValues(); const refColIndex = 0; // columna A = "Cart Ref" const emailColIndex = 1; // columna B = "Email" const convertidoColIndex = 8; // columna I = "Convertido" let matches = 0; for (let i = 1; i < values.length; i++) { const matchByRef = data.cart_ref && values[i][refColIndex] === data.cart_ref; const matchByEmail = !data.cart_ref && data.email && values[i][emailColIndex] === data.email && !values[i][convertidoColIndex]; if (matchByRef || matchByEmail) { sheet.getRange(i + 1, convertidoColIndex + 1).setValue("Sí — " + new Date().toISOString()); matches++; } } return jsonOut({ status: matches > 0 ? "ok" : "not_found", action: "mark_converted", rows: matches }); }
